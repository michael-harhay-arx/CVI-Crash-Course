\chapter{Tips \& Tricks }
\hypertarget{tips}{}\label{tips}\index{Tips \& Tricks@{Tips \& Tricks}}
The following tips and tricks are helpful to know.\hypertarget{tips_gentips}{}\doxysection{\texorpdfstring{General Tips \& Tricks}{General Tips \& Tricks}}\label{tips_gentips}

\begin{DoxyItemize}
\item {\bfseries{Help Menu}}:\+ Press {\bfseries{F1}} to access the CVI help documentation.
\item {\bfseries{Generate Prototype}}:\+ Select {\bfseries{Build \texorpdfstring{$>$}{>} Generate Prototypes}} to generate a file containing declarations for global and static functions, as well as external declarations for global variables. These can be copied into your existing source and header files.
\item {\bfseries{Tags}}:\+ Select a line and press {\bfseries{Shift+\+F2}} to "{}tag"{} or "{}bookmark"{} that line. Then, press {\bfseries{F2}} to cycle through all tagged lines.
\end{DoxyItemize}\hypertarget{tips_hotkeys}{}\doxysection{\texorpdfstring{Hotkeys}{Hotkeys}}\label{tips_hotkeys}

\begin{DoxyItemize}
\item {\bfseries{Comment Selection}}:\+ Select code to comment out, press {\bfseries{Ctrl+\+F3}}
\item {\bfseries{Uncomment Selection}}:\+ Select code to uncomment, press {\bfseries{Ctrl+\+F4}}
\item {\bfseries{Show Function Prototype}}:\+ Place cursor in function, press {\bfseries{Ctrl+\+Shift+\+Space}}
\item {\bfseries{Find}}:\+ {\bfseries{Ctrl+F}}
\item {\bfseries{Replace}}:\+ {\bfseries{Ctrl+H}}
\item {\bfseries{Go to line}}:\+ {\bfseries{Ctrl+L}}
\item {\bfseries{Build project}}:\+ {\bfseries{Ctrl+M}}
\item {\bfseries{Build specific file within project}}:\+ {\bfseries{Ctrl+K}}
\item {\bfseries{Go to line}}:\+ Select item to check defition of, press {\bfseries{Ctrl+I}}
\item {\bfseries{Go back}}:\+ After going to definition via {\bfseries{Ctrl+I}}, press {\bfseries{Ctrl+G}} to go back 
\end{DoxyItemize}
